try{const v=localStorage.getItem('tfDxN2');if(v)S={...S,...JSON.parse(v)};}catch(e){}

function save(){try{localStorage.setItem('tfDxN2',JSON.stringify(S));}catch(e){}}


// ===== VIEWS =====

document.querySelectorAll('.main-tab').forEach(t=>{t.addEventListener('click',function(){

  this.classList.add('active');

  document.getElementById('view-'+this.dataset.view).classList.add('active');

  if(this.dataset.view==='progress')renderProgress();

});});

// ===== CYCLE CHECK =====

function checkCycle(){

  let all=true;

  for(let d=1;d<=8;d++){P[d].items.forEach((_,i)=>{if(!S.chk[`${d}-${i}`])all=false;});}

  if(all&&P[1].items.length>0){

    let tot=0,dn=0,perf=0;

    for(let d=1;d<=8;d++){let dp=true;P[d].items.forEach((_,i)=>{tot++;if(S.chk[`${d}-${i}`])dn++;else dp=false;});if(dp)perf++;}

    document.getElementById('cPct').textContent=Math.round(dn/tot*100)+'%';

    document.getElementById('cDays').textContent=perf;

    document.getElementById('nxtC').textContent=S.cycle+1;

    document.getElementById('mCycle').classList.add('show');

  }

}


function newCycle(){

  let tot=0,dn=0;

  for(let d=1;d<=8;d++){P[d].items.forEach((_,i)=>{tot++;if(S.chk[`${d}-${i}`])dn++;});}

  S.cHist.push({c:S.cycle,pct:Math.round(dn/tot*100),date:Date.now()});

  S.cycle++;S.chk={};S.day=1;save();

  document.getElementById('mCycle').classList.remove('show');

  document.getElementById('cycleBanner').classList.add('show');

  document.getElementById('cycleNumB').textContent='C'+S.cycle;

  setTimeout(()=>document.getElementById('cycleBanner').classList.remove('show'),4000);

  render();

}


document.getElementById('startCycle').addEventListener('click',newCycle);


// ===== RENDER =====

function render(){renderDays();renderTL();renderMetrics();updCycle();}


function updCycle(){

  document.getElementById('cycleBadge').textContent='Cycle '+S.cycle;

  document.getElementById('nCyc').textContent='C'+S.cycle;

  document.getElementById('pCyc').textContent='Cycle '+S.cycle;

}


function renderDays(){

  const el=document.getElementById('daySel');el.innerHTML='';

  for(let i=1;i<=8;i++){

    const c=document.createElement('div');c.className=`day-chip ${i===S.day?'active':''}`;

    c.innerHTML=`<div class="num">J${i}</div><div class="typ">${P[i].type}</div>`;

    c.addEventListener('click',()=>{S.day=i;save();render();});el.appendChild(c);

  }

  document.getElementById('dayTitle').textContent=`Jour ${S.day} — ${P[S.day].type}`;

  document.getElementById('dayBadge').textContent=P[S.day].badge;

}


let eIdx=null;

function renderTL(){

  const tl=document.getElementById('timeline');const day=P[S.day];

  const items=S.filter==='all'?day.items:day.items.filter(x=>x.c===S.filter);

  tl.innerHTML='';

  items.forEach(item=>{

    const ri=day.items.indexOf(item);const key=`${S.day}-${ri}`;const done=S.chk[key];

    const div=document.createElement('div');div.className=`tl-item ${done?'done':''}`;

    div.innerHTML=`<div class="tl-time">${item.t}</div><div class="tl-check ${done?'on':''}" data-k="${key}"></div><div class="tl-icon ${item.c}">${item.i}</div><div class="tl-content"><h4>${item.a}</h4><p>${item.d}</p></div><button class="tl-edit" data-i="${ri}">✏️</button>`;

    div.querySelector('.tl-check').addEventListener('click',function(){const k=this.dataset.k;S.chk[k]=!S.chk[k];if(!S.chk[k])delete S.chk[k];save();render();checkCycle();});

    div.querySelector('.tl-edit').addEventListener('click',function(){eIdx=parseInt(this.dataset.i);const it=P[S.day].items[eIdx];document.getElementById('eTime').value=it.t;document.getElementById('eAct').value=it.a;document.getElementById('eDet').value=it.d;document.getElementById('mEdit').classList.add('show');});

    tl.appendChild(div);

  });

}


function renderMetrics(){

  const day=P[S.day];const tot=day.items.length;

  let dn=0,mD=0,mT=0,sD=0,sT=0,wD=0,wT=0;

  day.items.forEach((item,idx)=>{const k=`${S.day}-${idx}`;if(S.chk[k])dn++;if(item.c==='meal'){mT++;if(S.chk[k])mD++;}if(item.c==='supplement'){sT++;if(S.chk[k])sD++;}if(item.c==='workout'){wT++;if(S.chk[k])wD++;}});

  const pct=tot>0?Math.round(dn/tot*100):0;

  document.getElementById('nDisc').textContent=pct+'%';

  document.getElementById('nR1').style.strokeDashoffset=94-(94*pct/100);

  document.getElementById('ringPct').textContent=pct+'%';

  document.getElementById('rM').style.strokeDashoffset=553-(553*(mT>0?mD/mT:0));

  document.getElementById('rS').style.strokeDashoffset=465-(465*(sT>0?sD/sT:0));

  document.getElementById('rW').style.strokeDashoffset=377-(377*(wT>0?wD/wT:0));

  document.getElementById('lM').textContent=`${mD}/${mT}`;

  document.getElementById('lS').textContent=`${sD}/${sT}`;

  document.getElementById('lW').textContent=`${wD}/${wT}`;

  document.getElementById('tlBadge').textContent=`${dn}/${tot}`;

  document.getElementById('discPct').textContent=pct+'%';

  const bl=document.getElementById('discBlocks');bl.innerHTML='';

  day.items.forEach((_,i)=>{const b=document.createElement('div');b.className=`disc-block ${S.chk[`${S.day}-${i}`]?'on':''}`;bl.appendChild(b);});

  let found=false;

  for(let i=0;i<day.items.length;i++){if(!S.chk[`${S.day}-${i}`]){document.getElementById('nextAct').textContent=`${day.items[i].a} — ${day.items[i].d.substring(0,35)}`;document.getElementById('nextTime').textContent=day.items[i].t;found=true;break;}}

  if(!found){document.getElementById('nextAct').textContent="✅ Journée complète !";document.getElementById('nextTime').textContent="🏆";}

  let strk=0;for(let d=1;d<=8;d++){let a=true;P[d].items.forEach((_,i)=>{if(!S.chk[`${d}-${i}`])a=false;});if(a)strk++;}

  const totStrk=strk+(S.cHist.length*8);

  document.getElementById('nStrk').textContent=totStrk+'🔥';

  document.getElementById('nR2').style.strokeDashoffset=94-(94*Math.min(strk/8,1));

  document.getElementById('streakBig').textContent=totStrk+' jours';

  document.getElementById('sW').textContent=S.stats.w?S.stats.w+' kg':'—';

  document.getElementById('sF').textContent=S.stats.f?S.stats.f+'%':'—';

  document.getElementById('sT').textContent=S.stats.t?S.stats.t+' cm':'—';

  document.getElementById('sA').textContent=S.stats.a?S.stats.a+' cm':'—';

}


function renderProgress(){

  const c1=document.getElementById('chart1');c1.innerHTML='';

  const cols=['#FF2D55','#0A84FF','#FF9F0A','#30D158','#BF5AF2','#64D2FF','#FFD60A','#FF375F'];

  for(let d=1;d<=8;d++){let dn=0;P[d].items.forEach((_,i)=>{if(S.chk[`${d}-${i}`])dn++;});const p=P[d].items.length>0?Math.round(dn/P[d].items.length*100):0;const w=document.createElement('div');w.className='chart-bar-wrap';w.innerHTML=`<div class="chart-bar" style="height:${Math.max(p,4)}%;background:${cols[d-1]}"></div><div class="chart-label">J${d}</div>`;c1.appendChild(w);}

  const c2=document.getElementById('chart2');c2.innerHTML='';

  [['meal','var(--green)','Repas'],['supplement','var(--purple)','Suppl.'],['workout','var(--blue)','Train.']].forEach(([cat,col,lbl])=>{let t=0,d=0;for(let dd=1;dd<=8;dd++){P[dd].items.forEach((it,i)=>{if(it.c===cat){t++;if(S.chk[`${dd}-${i}`])d++;}});}const p=t>0?Math.round(d/t*100):0;const w=document.createElement('div');w.className='chart-bar-wrap';w.innerHTML=`<div class="chart-bar" style="height:${Math.max(p,4)}%;background:${col};width:45px"></div><div class="chart-label">${lbl}</div>`;c2.appendChild(w);});

  let rM=0,rS=0,rW=0,rC=0;

  for(let d=1;d<=8;d++){P[d].items.forEach((it,i)=>{if(S.chk[`${d}-${i}`]){if(it.c==='meal')rM++;if(it.c==='supplement')rS++;if(it.c==='workout'){rW++;if(it.a.includes('Cardio'))rC++;}}});}

  document.getElementById('rpM').textContent=rM;document.getElementById('rpS').textContent=rS;document.getElementById('rpW').textContent=rW;document.getElementById('rpC').textContent=rC;

}
